/**
 * Ruling 1 (2026-08-15). Execution-gap questions dedupe per SOURCE LOG, permanently —
 * Monday's and Wednesday's spray logs may BOTH be asked for a dose. Context questions
 * (crop stage, previous observation, learning) keep day-based cooldowns, because they
 * build the relationship rather than repair one record.
 *
 * The split is read from bank metadata already present: GAP_LENS assigns
 * WHAT/DOSE/SCOPE/CARRIER/COST to 'Execution'. No new vocabulary.
 */
function isPerLogScoped(q: DfesQuestion): boolean {
    return q.triggerType === 'Gap' && q.lens === 'Execution';
}

/**
 * Already asked for THIS log — permanent, no time window. Ruling 1 says "the same log
 * must never receive the same question twice"; that has no time component, so this is
 * deliberately NOT expressed as an age comparison.
 *
 * Falls back to false when either side lacks a log id: every question_events row written
 * before 3.1 has daily_log_id NULL, and treating those as "asked for no log" would
 * unblock every gap question at once.
 */
function askedForLog(q: DfesQuestion, recent: RecentQuestionEvent[], sourceLogId?: string): boolean {
    if (!sourceLogId) return false;
    return recent.some(e => e.questionKey === q.questionKey && e.dailyLogId === sourceLogId);
}

function eligible(
    q: DfesQuestion | undefined,
    recent: RecentQuestionEvent[],
    sourceLogId?: string,
): q is DfesQuestion {
    if (!approved(q)) return false;
    if (isPerLogScoped(q)) {
        // Per-log questions: permanent exclusion for this log, and the day cooldown
        // still applies to rows that predate per-log tracking.
        if (askedForLog(q, recent, sourceLogId)) return false;
        const legacyOnly = recent.filter(e => e.dailyLogId === null);
        return !onCooldown(q, legacyOnly);
    }
    return !onCooldown(q, recent);
}

// src/apps/ShramSafal/ShramSafal.Domain/Dfes/ProductWaterAffinity.cs
namespace ShramSafal.Domain.Dfes;

/// <summary>
/// FOUNDER DECISION 14 (2026-08-16). Whether an input product owes the water/carrier
/// question — decided from the PRODUCT, never from inputs[].method. A dry granular
/// (DAP / urea / MOP) carries no water; a water-soluble fertiliser (0:52:34 MKP) does,
/// whatever method flag rides along; an unrecognised product keeps asking (P4).
/// </summary>
public enum WaterAffinity { WaterCarried, Dry, Unknown }

public static class ProductWaterAffinity
{
    private static readonly System.Text.RegularExpressions.Regex GradeShape =
        new(@"^\s*\d{1,2}[-:]\d{1,2}[-:]\d{1,2}\s*$",
            System.Text.RegularExpressions.RegexOptions.Compiled);

    // Broadcast/soil granulars. Deliberately a SHORT, named list: an unknown name must
    // fall to Unknown and keep asking, never be guessed dry.
    private static readonly HashSet<string> DryGranulars =
        new(StringComparer.OrdinalIgnoreCase)
        { "DAP", "urea", "MOP", "FYM", "SSP", "MOP potash", "potash", "single super phosphate" };

    /// <summary>Resolution order is fixed and total.</summary>
    public static WaterAffinity Resolve(string? npkGrade, string? productName)
    {
        // 1. A recognised NPK grade is water-soluble BY DEFINITION. This is what makes
        //    "0 52 34" self-classifying with no flag.
        var grade = Canonical(npkGrade) ?? Canonical(productName);
        if (grade is not null && NpkGradeTable.KnownGrades.ContainsKey(grade))
            return WaterAffinity.WaterCarried;

        // 2. A recognised grape input: use its agronomic role.
        var role = GrapeProductRoles.Entries
            .FirstOrDefault(e => Matches(e, productName))?.AgronomicRole;
        if (role is not null)
            return role.Contains("paste", StringComparison.OrdinalIgnoreCase)
                ? WaterAffinity.Dry           // painted, not sprayed
                : WaterAffinity.WaterCarried; // fungicide / insecticide / PGR / foliar

        // 3. A named dry granular.
        if (!string.IsNullOrWhiteSpace(productName) && DryGranulars.Contains(productName.Trim()))
            return WaterAffinity.Dry;

        // 4. Anything else. P4 — do not guess.
        return WaterAffinity.Unknown;
    }

    private static string? Canonical(string? s)
        => !string.IsNullOrWhiteSpace(s) && GradeShape.IsMatch(s) ? s.Trim().Replace(':', '-') : null;

    private static bool Matches(GrapeProduct e, string? productName) =>
        !string.IsNullOrWhiteSpace(productName)
        && (string.Equals(e.CanonicalName, productName.Trim(), StringComparison.OrdinalIgnoreCase)
            || e.SttAliases.Contains(productName.Trim(), StringComparer.OrdinalIgnoreCase)
            || e.DevanagariAliases.Contains(productName.Trim(), StringComparer.OrdinalIgnoreCase));
}

### The design decision this task must record

**`ssf.question_events` is append-only by privilege** (`REVOKE UPDATE, DELETE`,
`20260713052440_AddDfesDataSpine.cs:256`). So a row written on **tap** can never afterwards acquire
the answer text. Writing on tap — which is what the old plan does — permanently guarantees
`response = NULL`. **Therefore the write is deferred until the outcome is known, and exactly one row
is ever written per (log, question).**

| Moment | What happens |
|---|---|
| Farmer taps the question card | Nothing is written. The selection (`questionKey`, `SelectedQuestion`, `shownAtUtc`, `sourceLogId`) is stashed by `pendingQuestionAnswer.ts`, and the app routes to the mic with the question pinned above it. |
| A parse comes back for that farm/day | One `recordQuestionEvent` with `skipped: false`, **`response` = the farmer's own transcript text**, `dailyLogId` = the stashed `sourceLogId`. Stash cleared. |
| He leaves without speaking (cancel / navigate away / next app open) | One `recordQuestionEvent` with `skipped: true`, `response: null`. Stash cleared. |
| Any retry of either | Idempotent — **3.3**'s `FindQuestionEventAsync` + `ux_question_events_log_question` return the existing row. |

stops only when a `recentEvents` row carries today's `createdAtLocalDate` (`:226-229`). While the
stash is live no row exists, so `useDfesQuestion` merges the stash in as a synthetic
`RecentQuestionEvent` (`createdAtLocalDate = todayLocalDate`, `dailyLogId = sourceLogId`) before
calling the selector. This is client-side only and never reaches the server.

### Decision: chips are `DisturbanceEvent`; the declaration is `dayOutcome`

- **The declaration itself is `dayOutcome`.** "There was no work today" is a statement about the
  **day**, not about a blockage. Forcing it into `disturbance_events` would set
  `ClassifierSignals.HasDisturbance` true for a plain rest day (`DfesLensExtractor.cs:128`) and
  return `reasonCode = "blocked"` instead of `"rest"` (`:334`) — recording the wrong fact, which is
  a P3/P8 violation, not a shortcut.
- **The reason chips are `DisturbanceEvent`.** They map 1:1 onto the existing `DisturbanceCause`
  enum (`machinery`, `electricity`, `weather`, `water_source`, `pest`, `disease`, `labour_shortage`,
  `material_shortage`, `other` — `DisturbanceCause.cs:5`), and `LedgerDerivationService.cs:342-364`
  already writes one from a `disturbance` object on the wire. Chips therefore need **no new table**.
- **Chips MUST be optional (doctrine P9).** `DisturbanceEvent.Create` requires a non-empty `reason`
  (`LedgerDerivationService.cs:346` skips the row without one), so a chip-less declaration simply
  writes **no** `DisturbanceEvent`. The record commits regardless. This is precisely why the
  declaration cannot live in `disturbance_events`: an optional field would be rejecting the record.
- **Non-speech fallback.** `globalSheets.tsx:62-64`'s existing `'no_work'` button (rendered by
  `QuickLogSheet.tsx:202-209`) becomes a real declaration: it saves a `DailyLog` with
  `dayOutcome: 'NO_WORK_PLANNED'`, provenance `manual`, and **no** buckets. Chips are offered after
  the save and are skippable. A farmer who cannot or will not speak declares his day with one tap.

**🛑 This adds a migration the MASTER change surface does not list.** MASTER §CHANGE SURFACE names
three new migrations (3.3, 3.12, Wave 4); this is a **fourth**. It is **additive and nullable** — one
`day_outcome text NULL` column on `ssf.daily_logs` — so `classify-migration.py` classifies it
`additive`, not destructive. There is no migration-free alternative: layers 4 and 5 are joined only
through the database, and the only existing table on that path is `disturbance_events`, which cannot
carry an optional declaration (see above). **Raise this with the founder before implementing.**
